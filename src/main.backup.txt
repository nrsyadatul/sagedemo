import React,{useEffect,useState} from 'react';
import {createRoot} from 'react-dom/client';
import {supabase,configured} from './supabase';
import './style.css';
const demo={mine:{balance:12450,spent:2180,budget:4000,goals:4700,goalTarget:14000,wishes:2700},partner:{balance:9800,spent:1540,budget:3000,goals:3900,goalTarget:12000,wishes:1800},shared:{balance:4600,spent:1120,budget:2500,goals:1800,goalTarget:6000,wishes:2400}};
const money=n=>'RM '+n.toLocaleString('en-MY',{maximumFractionDigits:0});
const greeting=()=>{const h=new Date().getHours();return h<12?'Good morning':h<17?'Good afternoon':'Good evening'};
const Progress=({value,max})=><div className="progress"><span style={{width:`${Math.min(100,100*value/max)}%`}}/></div>;
function App(){
 const [session,setSession]=useState(null),[loading,setLoading]=useState(configured),[tab,setTab]=useState('home'),[view,setView]=useState('mine'),[authMode,setAuthMode]=useState('password'),[email,setEmail]=useState(''),[password,setPassword]=useState(''),[name,setName]=useState(''),[profile,setProfile]=useState(null),[household,setHousehold]=useState(null),[inviteMethod,setInviteMethod]=useState('code'),[inviteCode,setInviteCode]=useState(''),[inviteEmail,setInviteEmail]=useState(''),[joinCode,setJoinCode]=useState(''),[householdName,setHouseholdName]=useState('Our Household'),[msg,setMsg]=useState(''),[busy,setBusy]=useState(false),[showAuth,setShowAuth]=useState(false),[showHouse,setShowHouse]=useState(false);
 const user=session?.user;
 async function refresh(userId){if(!supabase||!userId)return;const [p,m]=await Promise.all([supabase.from('profiles').select('display_name').eq('id',userId).maybeSingle(),supabase.from('household_members').select('household_id,households(id,name)').eq('user_id',userId).limit(1).maybeSingle()]);if(p.error)setMsg(p.error.message);setProfile(p.data);setName(p.data?.display_name||'');setHousehold(m.data?.households||null)}
 useEffect(()=>{if(!supabase)return;supabase.auth.getSession().then(({data})=>{setSession(data.session);setLoading(false);if(data.session)refresh(data.session.user.id)});const {data:{subscription}}=supabase.auth.onAuthStateChange((_event,s)=>{setSession(s);if(s)setTimeout(()=>refresh(s.user.id),0);else{setProfile(null);setHousehold(null)}});return()=>subscription.unsubscribe()},[]);
 async function action(fn){setBusy(true);setMsg('');try{const result=await fn();if(result?.error)throw result.error;if(result?.data?.message)setMsg(result.data.message);return result}catch(e){setMsg(e.message||'Something went wrong');return null}finally{setBusy(false)}}
 const auth=async(signUp=false)=>action(async()=>{let r;if(authMode==='link')r=await supabase.auth.signInWithOtp({email,options:{emailRedirectTo:window.location.origin}});else if(signUp)r=await supabase.auth.signUp({email,password,options:{emailRedirectTo:window.location.origin}});else r=await supabase.auth.signInWithPassword({email,password});if(!r.error&&(!r.data?.session))setMsg('Check your email to continue.');return r});
 const saveProfile=()=>action(async()=>{const r=await supabase.from('profiles').upsert({id:user.id,display_name:name.trim()});if(!r.error)await refresh(user.id);return r});
 const createHouse=()=>action(async()=>{const r=await supabase.rpc('create_household',{household_name:householdName});if(!r.error)await refresh(user.id);return r});
 const joinHouse=()=>action(async()=>{const r=await supabase.rpc('accept_household_invite',{raw_code:joinCode.trim()});if(!r.error){setMsg('Household joined successfully.');await refresh(user.id)}return r});
 const makeInvite=()=>action(async()=>{const r=await supabase.rpc('create_household_invite',{target_email:inviteMethod==='email'?inviteEmail.trim():null});if(!r.error){setInviteCode(r.data);setMsg(inviteMethod==='email'?'Invite created. Copy and share the code securely with the intended email recipient. Automatic email delivery requires a separate server-side email integration.':'One-time code generated. Share it securely with your partner.')}return r});
 const d=demo[view],privatePartner=view==='partner';
 const card=(children)=><div className="card">{children}</div>;
 const head=(title,link,fn)=><div className="section-head"><h2>{title}</h2>{link&&<button onClick={fn}>{link} →</button>}</div>;
 const home=<><div className="hero"><small>{view==='mine'?'Total balance':privatePartner?'Partner balance summary':'Shared balance'}</small><div className="balance">{money(d.balance)}</div><div className="hero-bottom"><div className="metric"><small>Monthly spending</small><strong>{money(d.spent)}</strong></div><div className="metric"><small>Budget remaining</small><strong>{money(d.budget-d.spent)}</strong></div></div></div>{head('Budget remaining','View budget',()=>setTab('budget'))}{card(<><div className="card-head"><strong>{money(d.budget-d.spent)} left this month</strong><span className="muted">{Math.round(d.spent/d.budget*100)}% used</span></div><Progress value={d.spent} max={d.budget}/></>)}{head('Savings goals','View goals',()=>setTab('goals'))}{card(<><div className="card-head"><strong>{money(d.goals)} saved</strong><span className="muted">{Math.round(d.goals/d.goalTarget*100)}%</span></div><Progress value={d.goals} max={d.goalTarget}/></>)}{head('Cards',null)}<div className="two-cards"><div className="card mini"><small>Debit · Available</small><strong>RM 2,450</strong></div><div className="card mini"><small>Credit · Used</small><strong>RM 1,280</strong></div></div><p className="muted">Card amounts are sample placeholders. Live tracking is reserved for Phase 3.</p>{head('Recent transactions','See all',()=>setMsg('Transaction history will be implemented in Phase 3.'))}{card(privatePartner?<div className="notice">Private transactions are hidden. Only summaries are available.</div>:<><div className="transaction"><span>🛒 &nbsp; Groceries</span><strong>− RM 125</strong></div><div className="transaction"><span>☕ &nbsp; Coffee</span><strong>− RM 18</strong></div></>)}{head('Wishlist preview','View wishlist',()=>setTab('wishlist'))}{card(<><div className="card-head"><strong>Wishlist</strong><span className="muted">{money(d.wishes)} planned</span></div><div className="wishlist"><div className="wish">👟<strong>Running shoes</strong><small>RM 450</small></div><div className="wish">✈️<strong>Weekend trip</strong><small>RM 1,500</small></div></div></>)}</>;
 const other=tab==='budget'?<><div className="hero"><small>Remaining this month</small><div className="balance">{money(d.budget-d.spent)}</div><small>Monthly category budgets</small></div>{[['Food & drinks',620,900],['Transport',280,500],['Shopping',350,600]].map(([n,s,l])=><div className="card" key={n}><div className="card-head"><strong>{n}</strong><span className="muted">{money(s)} / {money(l)}</span></div><Progress value={s} max={l}/></div>)}</>:tab==='goals'?<><div className="hero"><small>Total saved towards goals</small><div className="balance">{money(d.goals)}</div><small>Manual contributions</small></div>{[['Emergency fund',3500,10000],['Holiday fund',1200,4000]].map(([n,s,l])=><div className="card" key={n}><div className="card-head"><strong>{n}</strong><span className="muted">{Math.round(100*s/l)}%</span></div><Progress value={s} max={l}/></div>)}</>:<><div className="hero"><small>Wishlist total</small><div className="balance">{money(d.wishes)}</div><small>Shopping cards</small></div><div className="wishlist"><div className="card wish-large">👟<strong>Running shoes</strong><small>RM 450</small></div><div className="card wish-large">✈️<strong>Weekend getaway</strong><small>RM 1,500</small></div></div></>;
 return <main className="app">{!configured&&<div className="banner"><strong>Preview mode</strong> · Add Supabase environment variables and run the SQL setup to enable real accounts. All financial figures are examples.</div>}{loading?<p>Loading Sage…</p>:<><header className="header"><div><div className="eyebrow">SAGE / {tab.toUpperCase()}</div><h1>{tab==='home'?`${greeting()}, ${profile?.display_name||'there'}`:({budget:'Your budget',goals:'Savings goals',wishlist:'Wishlist'})[tab]}</h1></div><button className="avatar" aria-label="Profile and household settings" onClick={()=>setShowAuth(true)}>◯</button></header><div className="switcher">{[['mine','Mine'],['partner','Partner'],['shared','Shared']].map(([id,label])=><button key={id} className={view===id?'active':''} onClick={()=>setView(id)}>{label}</button>)}</div>{tab==='home'?home:other}<div className="preview">Sage Phase 2 · Finance figures are sample data, not live balances</div></>}{showAuth&&<div className="overlay" role="dialog" aria-modal="true" aria-label="Account settings"><div className="modal"><button className="close" onClick={()=>{setShowAuth(false);setMsg('')}}>×</button><h2>{user?'Your Sage account':'Welcome to Sage'}</h2>{!configured?<div className="notice">Cloud sign-in is not configured. Set your Supabase URL and publishable key to enable Phase 2.</div>:!user?<><div className="switcher"><button className={authMode==='password'?'active':''} onClick={()=>setAuthMode('password')}>Password</button><button className={authMode==='link'?'active':''} onClick={()=>setAuthMode('link')}>Email link</button></div><label>Email<input type="email" value={email} onChange={e=>setEmail(e.target.value)} autoComplete="email"/></label>{authMode==='password'&&<label>Password<input type="password" value={password} onChange={e=>setPassword(e.target.value)} autoComplete="current-password"/></label>}<button className="primary" disabled={busy||!email} onClick={()=>auth(false)}>{authMode==='link'?'Send sign-in link':'Sign in'}</button>{authMode==='password'&&<button className="secondary" disabled={busy||!email||!password} onClick={()=>auth(true)}>Create account</button>}</>:<><p className="muted">Signed in as {user.email}</p><label>Display name<input value={name} onChange={e=>setName(e.target.value)}/></label><button className="primary" disabled={busy||!name.trim()} onClick={saveProfile}>Save profile</button><button className="secondary" onClick={()=>setShowHouse(true)}>Manage household</button><button className="text-button" onClick={()=>supabase.auth.signOut()}>Sign out</button></>}{msg&&<div className="notice" role="status">{msg}</div>}</div></div>}{showHouse&&user&&<div className="overlay top" role="dialog" aria-modal="true" aria-label="Household"><div className="modal"><button className="close" onClick={()=>setShowHouse(false)}>×</button><h2>Household sharing</h2>{household?<><p>Your household: <strong>{household.name}</strong></p><div className="switcher"><button className={inviteMethod==='code'?'active':''} onClick={()=>setInviteMethod('code')}>Invite code</button><button className={inviteMethod==='email'?'active':''} onClick={()=>setInviteMethod('email')}>Email invite</button></div>{inviteMethod==='email'&&<label>Partner email<input type="email" value={inviteEmail} onChange={e=>setInviteEmail(e.target.value)}/></label>}<button className="primary" disabled={busy||(inviteMethod==='email'&&!inviteEmail)} onClick={makeInvite}>Generate invitation</button>{inviteCode&&<div className="notice"><strong>Invite code: {inviteCode}</strong><button className="secondary" onClick={()=>navigator.clipboard?.writeText(inviteCode)}>Copy code</button></div>}</>:<><label>Household name<input value={householdName} onChange={e=>setHouseholdName(e.target.value)}/></label><button className="primary" disabled={busy||!householdName.trim()} onClick={createHouse}>Create household</button><p className="muted">Or join an existing household:</p><label>Invite code<input value={joinCode} onChange={e=>setJoinCode(e.target.value)}/></label><button className="secondary" disabled={busy||!joinCode.trim()} onClick={joinHouse}>Join household</button></>}{msg&&<div className="notice">{msg}</div>}</div></div>}<div className="nav-wrap"><nav className="nav"><div className="tabs">{[['home','⌂','Home'],['budget','▤','Budget'],['goals','◎','Goals'],['wishlist','♧','Wishlist']].map(([id,icon,label])=><button key={id} className={tab===id?'active':''} onClick={()=>{setTab(id);window.scrollTo(0,0)}}><span className="ico">{icon}</span><span className="txt">{label}</span></button>)}</div><button className="add" aria-label="Quick Add" onClick={()=>setMsg('Quick Add financial recording is planned for Phase 3.')}>+</button></nav></div>{msg&&!showAuth&&!showHouse&&<div className="toast" role="status" onClick={()=>setMsg('')}>{msg} ×</div>}</main>
}
createRoot(document.getElementById('root')).render(<App/>);
